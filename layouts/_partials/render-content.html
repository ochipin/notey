Ordinary Markdown headings retain their IDs, keeping Hugo's TOC intact; unique shortcode headings (including explicit IDs) retain theirs too. */ -}} {{- $html := printf "%s" .Content -}} {{- $headings $id true -}} {{- end -}} {{- /* Reserve every original spelling, so an added suffix can never steal a later heading's intentional ID. */ -}} {{- $originals := newScratch -}} {{- range findRE ` id="[^"]+"` $html -}} {{- $id := strings.TrimSuffix `"` (strings.TrimPrefix ` id="` .) | htmlUnescape -}} {{- $originals.Set $id true -}} {{- end -}} {{- $anchors := newScratch -}} {{- range $headings | htmlUnescape -}} {{- $anchor := $original -}} {{- if $used.Get $anchor -}} {{- $base := printf "%s--notey-%s" $original $scope -}} {{- $anchor = $base -}} {{- if or ($used.Get $anchor)
